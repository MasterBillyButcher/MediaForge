   374→    @CheckResult
   375→    fun getCookieListFromDatabase(): Result<List<Cookie>> = runCatching {
   376→        CookieManager.getInstance().run {
   377→            if (!hasCookies()) throw Exception("There is no cookies in the database!")
   378→            flush()
   379→        }
   380→        SQLiteDatabase.openDatabase(
   381→                context.dataDir.resolve("app_webview/Default/Cookies").absolutePath,
   382→                null,
   383→                OPEN_READONLY,
   384→            )
   385→            .run {
   386→                val projection =
   387→                    arrayOf(
   388→                        CookieScheme.HOST,
   389→                        CookieScheme.EXPIRY,
   390→                        CookieScheme.PATH,
   391→                        CookieScheme.NAME,
   392→                        CookieScheme.VALUE,
   393→                        CookieScheme.SECURE,
   394→                    )
   395→                val cookieList = mutableListOf<Cookie>()
   396→                query("cookies", projection, null, null, null, null, null).run {
   397→                    while (moveToNext()) {
   398→                        val expiry = getLong(getColumnIndexOrThrow(CookieScheme.EXPIRY))
   399→                        val name = getString(getColumnIndexOrThrow(CookieScheme.NAME))
   400→                        val value = getString(getColumnIndexOrThrow(CookieScheme.VALUE))
   401→                        val path = getString(getColumnIndexOrThrow(CookieScheme.PATH))
   402→                        val secure = getLong(getColumnIndexOrThrow(CookieScheme.SECURE)) == 1L
   403→                        val hostKey = getString(getColumnIndexOrThrow(CookieScheme.HOST))
   404→
   405→                        val host = if (hostKey[0] != '.') ".$hostKey" else hostKey
   406→                        cookieList.add(
   407→                            Cookie(
   408→                                domain = host,
   409→                                name = name,
   410→                                value = value,
   411→                                path = path,
   412→                                secure = secure,
   413→                                expiry = expiry,
   414→                            )
   415→                        )
   416→                    }
   417→                    close()
   418→                }
   419→                close()
   420→                cookieList
   421→            }
   422→    }

   458→                    subtitleLanguage
   459→                        .takeIf { it.isNotEmpty() }
   460→                        ?.let { addOption("--sub-langs", it) }
   461→                    if (embedSubtitle) {
   462→                        addOption("--embed-subs")
   463→                        if (keepSubtitle) {
   464→                            addOption("--write-subs")
   465→                        }
   466→                    } else {
   467→                        addOption("--write-subs")
   468→                    }
   469→                    when (convertSubtitle) {
   470→                        CONVERT_ASS -> addOption("--convert-subs", "ass")
   471→                        CONVERT_SRT -> addOption("--convert-subs", "srt")

   476→                }
   477→                if (mergeToMkv) {
   478→                    addOption("--remux-video", "mkv")

   480→                }
   481→                if (embedThumbnail) {
   482→                    addOption("--embed-thumbnail")
   483→                }
   484→                if (videoClips.isEmpty()) addOption("--embed-chapters")
   485→            }
   486→        }

   488→    @CheckResult
   489→    private fun DownloadPreferences.toAudioFormatSorter(): String =
   490→        this.run {
   491→            if (!useCustomAudioPreset) return@run ""
   492→            val format =
   493→                when (audioFormat) {
   494→                    M4A -> "acodec:aac"
   495→                    OPUS -> "acodec:opus"
   496→                    else -> ""
   497→                }
   498→            val quality =
   499→                when (audioQuality) {
   500→                    HIGH -> "abr~192"
   501→                    MEDIUM -> "abr~128"
   502→                    LOW -> "abr~64"
   503→                    else -> ""
   504→                }
   505→            return@run connectWithDelimiter(format, quality, delimiter = ",")
   506→        }
   507→
   508→    @CheckResult
   509→    private fun DownloadPreferences.toVideoFormatSorter(): String =
   510→        this.run {
   511→            val format =
   512→                when (videoFormat) {
   513→                    FORMAT_COMPATIBILITY -> "proto,vcodec:h264,ext"
   514→                    FORMAT_QUALITY ->
   515→                        if (supportAv1HardwareDecoding) {
   516→                            "vcodec:av01"
   517→                        } else {
   518→                            "vcodec:vp9.2"
   519→                        }
   520→
   521→                    else -> ""
   522→                }
   523→            val res =
   524→                when (videoResolution) {
   525→                    1 -> "res:2160"
   526→                    2 -> "res:1440"
   527→                    3 -> "res:1080"
   528→                    4 -> "res:720"
   529→                    5 -> "res:480"
   530→                    6 -> "res:360"
   531→                    7 -> "+res"
   532→                    else -> ""
   533→                }
   534→            val sorter = if (videoFormat == FORMAT_COMPATIBILITY) {
   535→                connectWithDelimiter(format, res, delimiter = ",")
   536→            } else {
   537→                connectWithDelimiter(res, format, delimiter = ",")
   538→            }
   539→            return@run sorter
   540→        }

   659→    @CheckResult
   660→    fun downloadVideo(
   661→        videoInfo: VideoInfo? = null,
   662→        playlistUrl: String = "",
   663→        playlistItem: Int = 0,
   664→        taskId: String,
   665→        downloadPreferences: DownloadPreferences,
   666→        progressCallback: ((Float, Long, String) -> Unit)?,
   667→    ): Result<List<String>> {
   668→        if (videoInfo == null)
   669→            return Result.failure(Throwable(context.getString(R.string.fetch_info_error_msg)))
   670→
   671→        with(downloadPreferences) {
   672→            val url =
   673→                playlistUrl.ifEmpty {
   674→                    videoInfo.originalUrl
   675→                        ?: videoInfo.webpageUrl
   676→                        ?: return Result.failure(
   677→                            Throwable(context.getString(R.string.fetch_info_error_msg))
   678→                        )
   679→                }
   680→            val request = YoutubeDLRequest(url)
   681→            val pathBuilder = StringBuilder()
   682→            val outputBuilder = StringBuilder()
   683→
   684→            request
   685→                .apply {
   686→                    addOption("--no-mtime")
   687→                    //                addOption("-v")
   688→                    if (cookies) {
   689→                        enableCookies(userAgentString)
   690→                    }
   691→                    if (restrictFilenames) {
   692→                        addOption("--restrict-filenames")
   693→                    }
   694→                    if (proxy) {
   695→                        enableProxy(proxyUrl)
   696→                    }
   697→                    if (forceIpv4) {
   698→                        addOption("-4")
   699→                    }
   700→                    if (debug) {
   701→                        addOption("-v")
   702→                    }
   703→                    if (useDownloadArchive) {
   704→                        val archiveFile = context.getArchiveFile()
   705→                        val archiveFileContent = archiveFile.readText()
   706→                        if (archiveFileContent.contains("${videoInfo.extractor} ${videoInfo.id}")) {
   707→                            return Result.failure(
   708→                                YoutubeDLException(
   709→                                    context.getString(R.string.download_archive_error)
   710→                                )
   711→                            )
   712→                        } else {
   713→                            useDownloadArchive()
   714→                        }
   715→                    }
   716→
   717→                    if (rateLimit && maxDownloadRate.isNumberInRange(1, 1000000)) {
   718→                        addOption("-r", "${maxDownloadRate}K")
   719→                    }
   720→
   721→                    if (playlistItem != 0 && downloadPlaylist) {
   722→                        addOption("--playlist-items", playlistItem)
   723→                        if (subdirectoryPlaylistTitle && !videoInfo.playlist.isNullOrEmpty()) {
   724→                            outputBuilder.append(PLAYLIST_TITLE_SUBDIRECTORY_PREFIX)
   725→                        }
   726→                        //                    addOption("--compat-options",
   727→                        // "no-youtube-unavailable-videos")
   728→                    } else {
   729→                        addOption("--no-playlist")
   730→                    }
   731→
   732→                    if (aria2c) {
   733→                        enableAria2c()
   734→                    } else if (concurrentFragments > 1) {
   735→                        addOption("--concurrent-fragments", concurrentFragments)
   736→                    }
   737→
   738→                    if (extractAudio || (videoInfo.vcodec == "none")) {
   739→                        if (privateDirectory) pathBuilder.append(App.privateDownloadDir)
   740→                        else pathBuilder.append(audioDownloadDir)
   741→                        addOptionsForAudioDownloads(
   742→                            id = videoInfo.id,
   743→                            preferences = downloadPreferences,
   744→                            playlistUrl = playlistUrl,
   745→                        )
   746→                    } else {
   747→                        if (privateDirectory) pathBuilder.append(App.privateDownloadDir)
   748→                        else pathBuilder.append(videoDownloadDir)
   749→                        addOptionsForVideoDownloads(downloadPreferences)
   750→                    }
   751→                    if (sponsorBlock) {
   752→                        addOption("--sponsorblock-remove", sponsorBlockCategory)
   753→                    }
   754→
   755→                    if (createThumbnail) {
   756→                        addOption("--write-thumbnail")
   757→                        addOption("--convert-thumbnails", "png")
   758→                    }
   759→                    if (subdirectoryExtractor) {
   760→                        pathBuilder.append("/${videoInfo.extractorKey}")
   761→                    }
   762→
   763→                    if (sdcard) {
   764→                        addOption("-P", context.getSdcardTempDir(videoInfo.id).absolutePath)
   765→                    } else {
   766→                        addOption("-P", pathBuilder.toString())
   767→                    }
   768→
   769→                    videoClips.forEach {
   770→                        addOption(
   771→                            "--download-sections",
   772→                            "*%d-%d".format(locale = Locale.US, it.start, it.end),
   773→                        )
   774→                    }
   775→                    if (newTitle.isNotEmpty()) {
   776→                        addCommands(listOf("--replace-in-metadata", "title", ".+", newTitle))
   777→                    }
   778→                    if (Build.VERSION.SDK_INT > 23 && !sdcard)
   779→                        addOption("-P", "temp:" + getExternalTempDir())
   780→
   781→                    if (splitByChapter) {
   782→                        addOption("-o", OUTPUT_TEMPLATE_CHAPTERS)
   783→                        addOption("--split-chapters")
   784→                    }
   785→
   786→                    val output =
   787→                        if (splitByChapter) {
   788→                            OUTPUT_TEMPLATE_SPLIT
   789→                        } else if (videoClips.isEmpty()) {
   790→                            outputTemplate
   791→                        } else {
   792→                            OUTPUT_TEMPLATE_CLIPS
   793→                        }
   794→
   795→                    addOption("-o", outputBuilder.append(output).toString())
   796→
   797→                    for (s in request.buildCommand()) Log.d(TAG, s)
   798→                }
   799→                .runCatching {
   800→                    YoutubeDL.getInstance()
   801→                        .execute(request = this, processId = taskId, callback = progressCallback)
   802→                }
   803→                .onFailure { th ->
   804→                    return if (
   805→                        sponsorBlock &&
   806→                            th.message?.contains("Unable to communicate with SponsorBlock API") ==
   807→                                true
   808→                    ) {
   809→                        th.printStackTrace()
   810→                        onFinishDownloading(
   811→                            preferences = this,
   812→                            videoInfo = videoInfo,
   813→                            downloadPath = pathBuilder.toString(),
   814→                            sdcardUri = sdcardUri,
   815→                        )
   816→                    } else Result.failure(th)
   817→                }
   818→            return onFinishDownloading(
   819→                preferences = this,
   820→                videoInfo = videoInfo,
   821→                downloadPath = pathBuilder.toString(),
   822→                sdcardUri = sdcardUri,
   823→            )
   824→        }
   825→    }

   827→    private fun onFinishDownloading(
   828→        preferences: DownloadPreferences,
   829→        videoInfo: VideoInfo,
   830→        downloadPath: String,
   831→        sdcardUri: String,
   832→    ): Result<List<String>> =
   833→        preferences.run {
   834→            val fileName =
   835→                preferences.newTitle.ifEmpty {
   836→                    videoInfo.filename
   837→                        ?: videoInfo.requestedDownloads?.firstOrNull()?.filename
   838→                        ?: videoInfo.title
   839→                }
   840→
   841→            Log.d(TAG, "onFinishDownloading: $fileName")
   842→            if (sdcard) {
   843→                moveFilesToSdcard(
   844→                        sdcardUri = sdcardUri,
   845→                        tempPath = context.getSdcardTempDir(videoInfo.id),
   846→                    )
   847→                    .onSuccess {
   848→                        if (privateMode) {
   849→                            return Result.success(emptyList())
   850→                        } else if (splitByChapter) {
   851→                            insertSplitChapterIntoHistory(videoInfo, it)
   852→                        } else {
   853→                            insertInfoIntoDownloadHistory(videoInfo, it)
   854→                        }
   855→                    }
   856→            } else {
   857→                FileUtil.scanFileToMediaLibraryPostDownload(
   858→                        title = fileName,
   859→                        downloadDir = downloadPath,
   860→                    )
   861→                    .run {
   862→                        if (privateMode) Result.success(emptyList())
   863→                        else
   864→                            Result.success(
   865→                                if (splitByChapter) {
   866→                                    insertSplitChapterIntoHistory(videoInfo, this)
   867→                                } else {
   868→                                    insertInfoIntoDownloadHistory(videoInfo, this)
   869→                                }
   870→                            )
   871→                    }
   872→            }
   873→        }

   876→    fun executeCustomCommandTask(
   877→        urlString: String,
   878→        taskId: String,
   879→        template: CommandTemplate,
   880→        preferences: DownloadPreferences,
   881→        progressCallback: ((Float, Long, String) -> Unit),
   882→    ): Result<YoutubeDLResponse> {
   883→        val urlList = urlString.split(Regex("[\n ]")).filter { it.isNotBlank() }
   884→
   885→        val request =
   886→            with(preferences) {
   887→                YoutubeDLRequest(urlList).apply {
   888→                    commandDirectory.takeIf { it.isNotEmpty() }?.let { addOption("-P", it) }
   889→                    addOption("--newline")
   890→                    if (aria2c) {
   891→                        enableAria2c()
   892→                    }
   893→                    if (useDownloadArchive) {
   894→                        useDownloadArchive()
   895→                    }
   896→                    if (restrictFilenames) {
   897→                        addOption("--restrict-filenames")
   898→                    }
   899→                    addOption(
   900→                        "--config-locations",
   901→                        FileUtil.writeContentToFile(template.template, context.getConfigFile())
   902→                            .absolutePath,
   903→                    )
   904→                    if (cookies) {
   905→                        enableCookies(userAgentString)
   906→                    }
   907→                }
   908→            }
   909→
   910→        return runCatching {
   911→            YoutubeDL.getInstance()
   912→                .execute(request = request, processId = taskId, callback = progressCallback)
   913→        }
   914→    }

   916→    suspend fun executeCommandInBackground(
   917→        url: String,
   918→        template: CommandTemplate = PreferenceUtil.getTemplate(),
   919→        downloadPreferences: DownloadPreferences = DownloadPreferences.createFromPreferences(),
   920→    ) {
   921→        downloadPreferences.run {
   922→            val taskId = Downloader.makeKey(url = url, templateName = template.name)
   923→            val notificationId = taskId.toNotificationId()
   924→            val urlList = url.split(Regex("[\n ]")).filter { it.isNotBlank() }
   925→
   926→            ToastUtil.makeToastSuspend(context.getString(R.string.start_execute))
   927→            val request =
   928→                YoutubeDLRequest(urlList).apply {
   929→                    commandDirectory.takeIf { it.isNotEmpty() }?.let { addOption("-P", it) }
   930→                    addOption("--newline")
   931→                    if (aria2c) {
   932→                        enableAria2c()
   933→                    }
   934→                    if (useDownloadArchive) {
   935→                        useDownloadArchive()
   936→                    }
   937→                    if (restrictFilenames) {
   938→                        addOption("--restrict-filenames")
   939→                    }
   940→                    addOption(
   941→                        "--config-locations",
   942→                        FileUtil.writeContentToFile(template.template, context.getConfigFile())
   943→                            .absolutePath,
   944→                    )
   945→                    if (cookies) {
   946→                        enableCookies(userAgentString)
   947→                    }
   948→                }
   949→
   950→            onProcessStarted()
   951→            withContext(Dispatchers.Main) { onTaskStarted(template, url) }
   952→            runCatching {
   953→                    val response =
   954→                        YoutubeDL.getInstance().execute(request = request, processId = taskId) {
   955→                            progress,
   956→                            _,
   957→                            text ->
   958→                            NotificationUtil.makeNotificationForCustomCommand(
   959→                                notificationId = notificationId,
   960→                                taskId = taskId,
   961→                                progress = progress.toInt(),
   962→                                templateName = template.name,
   963→                                taskUrl = url,
   964→                                text = text,
   965→                            )
   966→                            Downloader.updateTaskOutput(
   967→                                template = template,
   968→                                url = url,
   969→                                line = text,
   970→                                progress = progress,
   971→                            )
   972→                        }
   973→                    onTaskEnded(template, url, response.out + "\n" + response.err)
   974→                }
   975→                .onFailure {
   976→                    it.printStackTrace()
   977→                    if (it is YoutubeDL.CanceledException) return@onFailure
   978→                    it.message.run {
   979→                        if (isNullOrEmpty()) onTaskEnded(template, url)
   980→                        else onTaskError(this, template, url)
   981→                    }
   982→                }
   983→            onProcessEnded()
   984→        }
   985→    }

   987→    private fun checkIfAv1HardwareAccelerated(): Boolean {
   988→        if (PreferenceUtil.containsKey(AV1_HARDWARE_ACCELERATED)) {
   989→            return AV1_HARDWARE_ACCELERATED.getBoolean()
   990→        } else {
   991→            val res =
   992→                if (Build.VERSION.SDK_INT < Build.VERSION_CODES.Q) {
   993→                    false
   994→                } else {
   995→                    MediaCodecList(MediaCodecList.REGULAR_CODECS).codecInfos.any { info ->
   996→                        info.supportedTypes.any { it.equals("video/av01", ignoreCase = true) } &&
   997→                            info.isHardwareAccelerated
   998→                    }
   999→                }
  1000→            AV1_HARDWARE_ACCELERATED.updateBoolean(res)
  1001→            return res
  1002→        }
